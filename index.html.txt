        let currentTab = 0;
        let readNews = JSON.parse(localStorage.getItem('read_news') || '{}');
        let unreadNews = JSON.parse(localStorage.getItem('unread_news') || '[]');

        // Auto clean old items (7 days read, 30 days unread)
        function cleanStorage() {
            let now = new Date().getTime();
            for(let id in readNews) {
                if(now - readNews[id] > 7 * 24 * 60 * 60 * 1000) delete readNews[id];
            }
            localStorage.setItem('read_news', JSON.stringify(readNews));
        }
        cleanStorage();

        function renderGrid() {
            const grid = document.getElementById('paperGrid');
            grid.innerHTML = '';
            papers.forEach((p, idx) => {
                grid.innerHTML += `
                    <div class="card" onclick="openPaper('${p.name}', '${p.url}')">
                        <img src="${p.icon}" onerror="this.src='https://via.placeholder.com/32'">
                        <span>${p.name}</span>
                        <div class="badge">নতুন</div>
                    </div>
                `;
            });
        }

        function switchTab(index) {
            currentTab = index;
            document.querySelectorAll('.tab').forEach((t, i) => {
                t.classList.toggle('active', i === index);
            });
            if(index === 0) renderGrid();
            else if(index === 1) renderNewsList("সর্বশেষ সংবাদ ফিড");
            else if(index === 2) renderNewsList("অপঠিত সংবাদ");
            else if(index === 3) renderNewsList("পছন্দের সংবাদ");
        }

        function openPaper(name, url) {
            document.getElementById('reader-source').innerText = name;
            document.getElementById('readerBody').innerHTML = `
                <div style="text-align:center; padding: 40px;">
                    <h3>${name}</h3>
                    <p style="margin: 15px 0; color: var(--text-secondary);">খবর লোড হচ্ছে...</p>
                    <a href="${url}" target="_blank" style="color: var(--primary-color); text-decoration: none; font-weight: bold;">সরাসরি মূল ওয়েবসাইটে যান ➔</a>
                </div>
            `;
            document.getElementById('reader-view').style.display = 'flex';
        }

        function closeReader() {
            document.getElementById('reader-view').style.display = 'none';
        }

        function renderNewsList(title) {
            const container = document.getElementById('mainContainer');
            container.innerHTML = `<h3 style="margin-bottom: 15px; font-size:16px;">${title}</h3><div class="news-list">
                <div class="news-item" onclick="markAsRead(this, 1)">
                    <div class="news-title">প্রধানমন্ত্রীর সাথে উচ্চপর্যায়ের বৈঠক অনুষ্ঠিত</div>
                    <div class="news-meta">প্রথম আলো • ৫ মিনিট আগে</div>
                </div>
                <div class="news-item" onclick="markAsRead(this, 2)">
                    <div class="news-title">আজ দেশের বাজারে স্বর্ণের নতুন দাম নির্ধারণ</div>

                </div>
            </div>`;
        }

        function markAsRead(el, id) {
            el.classList.add('read');
            readNews[id] = new Date().getTime();
            localStorage.setItem('read_news', JSON.stringify(readNews));
        }

        function filterPapers() {
            let query = document.getElementById('searchBox').value.toLowerCase();
            const grid = document.getElementById('paperGrid');
            grid.innerHTML = '';
            papers.filter(p => p.name.toLowerCase().includes(query)).forEach(p => {
                grid.innerHTML += `
                    <div class="card" onclick="openPaper('${p.name}', '${p.url}')">
                        <img src="${p.icon}" onerror="this.src='https://via.placeholder.com/32'">
                        <span>${p.name}</span>
                    </div>
                `;
            });
        }

        // Swipe Gestures Logic
        let touchstartX = 0;
        let touchendX = 0;

        function handleGesture() {
            if (touchendX < touchstartX - 70) {
                // Swipe Left -> Next Tab
                if(currentTab < 3) switchTab(currentTab + 1);
            }
            if (touchendX > touchstartX + 70) {
                // Swipe Right -> Prev Tab
                if(currentTab > 0) switchTab(currentTab - 1);
            }
        }

        document.addEventListener('touchstart', e => { touchstartX = e.changedTouches[0].screenX; });
        document.addEventListener('touchend', e => { 
            touchendX = e.changedTouches[0].screenX; 
            handleGesture(); 
        });
